53## Flip a Coin! ##
54fac() {
55	local LANDED
56	LANDED=$(shuf -n1 -e "heads" "tails")
57	if [ "$1" == "$LANDED" ]; then
58		echo -e "\e[7m$LANDED"
59	else
60		echo "$LANDED"
61	fi
62}

64extract() {
65	if [ -f "$1" ]; then
66		case "$1" in
67		*.tar.bz2) tar xjf "$1" ;;
68		*.tar.gz) tar xzf "$1" ;;
69		*.bz2) bunzip2 "$1" ;;
70		*.rar) unrar x "$1" ;;
71		*.gz) gunzip "$1" ;;
72		*.tar) tar xf "$1" ;;
73		*.tbz2) tar xjf "$1" ;;
74		*.tgz) tar xzf "$1" ;;
75		*.zip) unzip "$1" ;;
76		*.Z) uncompress "$1" ;;
77		*.7z) 7z x "$1" ;;
78		*.zst) zstd -d "$1" ;;
79		*) echo "Unknown archive format" ;;
80		esac
81	else
82		echo "$1 is not a valid file"
83	fi
84}

86## cd to directory when exiting yazi with Q ##
87y() {
88#	local tmp
89	local tmp=$(mktemp -t "yazi-cwd.XXXXXX") cwd
90	command yazi "$@" --cwd-file="$tmp"
91	IFS= read -r -d '' cwd < "$tmp"
92	[ "$cwd" != "$PWD" ] && [ -d "$cwd" ] && builtin cd -- "$cwd"
93	rm -f -- "$tmp"
94}
